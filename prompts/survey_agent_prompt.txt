   ```javascript
   const WebSocket = typeof require !== "undefined" ? require("ws") : globalThis.WebSocket;
   const targets = await (await fetch("http://127.0.0.1:<PORT>/cdp/json")).json();
   const t = targets.find(x => x.type === "page" && !x.url.startsWith("chrome://") && x.url !== "about:blank") || targets.find(x => x.type === "page");
   const wsUrl = t.webSocketDebuggerUrl.replace(/ws:\/\/[^/]+/, "ws://127.0.0.1:<PORT>/cdp");
   const ws = new WebSocket(wsUrl);
   await new Promise((resolve, reject) => {
     ws.onopen = resolve; ws.onerror = reject;
     if (ws.on) { ws.on("open", resolve); ws.on("error", reject); }
   });

   let msgId = 0;
   const pending = new Map();
   const onMsg = (raw) => {
     const m = JSON.parse(typeof raw === "string" ? raw : (raw.data || raw.toString()));
     if (m.id && pending.has(m.id)) {
       const [resolve, reject] = pending.get(m.id);
       pending.delete(m.id);
       if (m.error) reject(m.error); else resolve(m.result);
     }
   };
   if (ws.on) ws.on("message", onMsg); else ws.onmessage = onMsg;

   const cdp = (method, params = {}) => new Promise((resolve, reject) => {
     const id = ++msgId;
     pending.set(id, [resolve, reject]);
     ws.send(JSON.stringify({ id, method, params }));
   });

   // Hello probe to confirm control before acting:
   const hello = await cdp("Runtime.evaluate", { expression: "document.title", returnByValue: true });
   nodeRepl.write(JSON.stringify({ ok: true, title: hello.result.value }));
   ```
