- [ ] **Step 4: Implement the shared deny boundary.** Add `loadRuntimeSafety(configPath, readFile = fs.readFileSync)` and `denyLiveExecution(safetyConfig)` in `scripts/live_execution_guard.mjs`. The loader catches read/parse failures and accepts only a JSON object with exactly one key, `live_execution_enabled`, whose value is boolean `false`; everything else becomes invalid. The deny function never returns `allowed: true`; a missing, malformed, or enabling config is denied as invalid, while the required false config is denied as disabled.

```js
import fs from "node:fs";

export function loadRuntimeSafety(configPath, readFile = fs.readFileSync) {
  try {
    const value = JSON.parse(readFile(configPath, "utf8"));
    if (!value || typeof value !== "object" || Array.isArray(value)) return null;
    if (Object.keys(value).length !== 1 || value.live_execution_enabled !== false) return null;
    return value;
  } catch {
    return null;
  }
}

export function denyLiveExecution(safetyConfig) {
  if (safetyConfig?.live_execution_enabled !== false) {
    return { allowed: false, reason: "safety_policy_missing_or_invalid" };
  }
  return { allowed: false, reason: "live_execution_disabled" };
}
```

Every caller must branch on `allowed` before invoking any live adapter. Do not add an alternate “live” mode in this task.
